# auto-pr Documentation Website Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deploy a Starlight docs site with custom landing page to GitHub Pages, sourcing content from the existing `docs/` directory.

**Architecture:** Isolated `website/` directory with its own Bun-managed dependencies. A TypeScript copy script transforms `docs/*.md` into Starlight-compatible content at build time (frontmatter injection, link rewriting, file renaming). CI deploys via artifact-based `actions/deploy-pages`.

**Tech Stack:** Astro 6.x, Starlight 0.38.x, Bun, GitHub Actions

### Task 2: Copy-docs script

**Files:**
- Create: `website/scripts/copy-docs.ts`

- [ ] **Step 1: Write the copy-docs script**

Create `website/scripts/copy-docs.ts`:

```ts
import { readdir, readFile, writeFile, mkdir, rm } from 'node:fs/promises';
import { join, basename, dirname } from 'node:path';

const REPO_ROOT = join(import.meta.dirname, '..', '..');
const DOCS_DIR = join(REPO_ROOT, 'docs');
const OUTPUT_DIR = join(import.meta.dirname, '..', 'src', 'content', 'docs');

interface DocMeta {
  order: number;
  label?: string;
}

const docsMeta: Record<string, DocMeta> = {
  'INTEGRATION.md': { order: 1, label: 'Getting Started' },
  'TROUBLESHOOTING.md': { order: 2 },
  'PR_TEMPLATE.md': { order: 3, label: 'PR Template Placeholders' },
  'ARCHITECTURE.md': { order: 10 },
  'CI.md': { order: 11, label: 'CI & Workflows' },
  'WORKFLOW_SECURITY.md': { order: 12 },
  'CII.md': { order: 13, label: 'CII Best Practices' },
};

const EXCLUDED_FILES = new Set(['README.md']);
const EXCLUDED_DIRS = new Set(['superpowers']);

function extractTitle(content: string): string {
  const match = content.match(/^#\s+(.+)$/m);
  return match ? match[1].trim() : 'Untitled';
}

function rewriteLinks(content: string): string {
  // Rewrite inter-doc links: (SOME_FILE.md) → (/auto-pr/some-file/)
  // Rewrite ADR links: (adr/0001-something.md) → (/auto-pr/adr/0001-something/)
  // Leave external URLs, anchors, and parent-relative links (../) untouched
  return content.replace(
    /\]\(([^)]+)\)/g,
    (full, href: string) => {
      // Skip external URLs, anchors, and parent-relative paths
      if (href.startsWith('http') || href.startsWith('#') || href.startsWith('../')) {
        return full;
      }

      // Handle anchor in link: FILE.md#section → /auto-pr/file/#section
      const [filePart, anchor] = href.split('#');
      if (!filePart || !filePart.endsWith('.md')) {
        return full;
      }

      // Convert filename to slug
      const slug = filePart
        .replace(/\.md$/, '')
        .toLowerCase()
        .replace(/_/g, '-');

      const anchorSuffix = anchor ? `#${anchor}` : '';
      return `](/auto-pr/${slug}/${anchorSuffix})`;
    },
  );
}

function buildFrontmatter(title: string, meta?: DocMeta): string {
  const lines = ['---', `title: "${title}"`];
  if (meta?.label) {
    lines.push(`sidebar:`);
    lines.push(`  label: "${meta.label}"`);
    lines.push(`  order: ${meta.order}`);
  } else if (meta) {
    lines.push(`sidebar:`);
    lines.push(`  order: ${meta.order}`);
  }
  lines.push('---', '');
  return lines.join('\n');
}

function stripH1(content: string): string {
  // Remove the first H1 heading since Starlight renders the title from frontmatter
  return content.replace(/^#\s+.+\n+/, '');
}

async function processFile(srcPath: string, destPath: string, meta?: DocMeta): Promise<void> {
  const raw = await readFile(srcPath, 'utf-8');
  const title = extractTitle(raw);
  const frontmatter = buildFrontmatter(title, meta);
  const rewritten = rewriteLinks(stripH1(raw));
  await mkdir(dirname(destPath), { recursive: true });
  await writeFile(destPath, frontmatter + rewritten);
}

async function main(): Promise<void> {
  // Clean output directory (except .gitkeep)
  const existing = await readdir(OUTPUT_DIR, { recursive: true }).catch(() => []);
  for (const entry of existing) {
    if (entry === '.gitkeep') continue;
    const full = join(OUTPUT_DIR, entry);
    await rm(full, { recursive: true, force: true }).catch(() => {});
  }

  // Process top-level docs
  const topLevel = await readdir(DOCS_DIR);
  for (const file of topLevel) {
    if (!file.endsWith('.md') || EXCLUDED_FILES.has(file)) continue;
    const destName = file.toLowerCase().replace(/_/g, '-');
    await processFile(
      join(DOCS_DIR, file),
      join(OUTPUT_DIR, destName),
      docsMeta[file],
    );
  }

  // Process ADRs
  const adrDir = join(DOCS_DIR, 'adr');
  const adrFiles = await readdir(adrDir);
  for (const file of adrFiles) {
    if (!file.endsWith('.md')) continue;
    if (file === 'adr-template.md') continue;
    const destName = file === 'README.md' ? 'index.md' : file.toLowerCase();
    await processFile(
      join(adrDir, file),
      join(OUTPUT_DIR, 'adr', destName),
    );
  }

  const topCount = topLevel.filter((f) => f.endsWith('.md') && !EXCLUDED_FILES.has(f)).length;
  const adrCount = adrFiles.filter((f) => f.endsWith('.md') && f !== 'adr-template.md').length;
  console.log(`Copied ${topCount} docs + ${adrCount} ADRs → ${OUTPUT_DIR}`);
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
```

### Task 5: CI workflow

**Files:**

- [ ] **Step 1: Create the deploy workflow**

Create `.github/workflows/deploy-pages.yml`:

```yaml
name: Deploy website

on:
  push:
    branches: [main]
    paths:
      - 'docs/**'
      - 'website/**'

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-24.04
    steps:
      - uses: actions/checkout@de0fac2e4500dabe0009e67214ff5f5447ce83dd # v6.0.2
      - uses: oven-sh/setup-bun@0c5077e51419868618aeaa5fe8019c62421857d6 # v2.2.0
      - run: bun install
        working-directory: website
      - run: bun run build
        working-directory: website
      - uses: actions/upload-pages-artifact@7b1f4a764d45c48632c6b24a0339c27f5614fb0b # v4.0.0
        with:
          path: website/dist

  deploy:
    needs: build
    runs-on: ubuntu-24.04
    environment:
      name: github-pages
      url: ${{ steps.deploy.outputs.page_url }}
    steps:
      - id: deploy
        uses: actions/deploy-pages@cd2ce8fcbc39b97be8ca5fce6e763baed58fa128 # v5.0.0
```
